\begin{frame}{hint 是什么：40 格的量子化}

    \end{itemize}
    \column{0.41\textwidth}

      \small $h[j]$ 回答："$w[j]$ 和 $w_a[j]$ 差了几个 682？"
      差 0 格 → $h=0$；差 3 格 → $h=3$。\\[0.3em]
      它是 $w-w_a$ 的\textbf{粗刻度}。

  \end{columns}

\begin{frame}{场定律：随机游走 + tent 核}
  \begin{columns}[T]
    \column{0.56\textwidth}
    \small
    $\varepsilon$-对齐把 $c_{\mathrm{signed}}=\varepsilon c_{\mathrm{can}}$
    里的 $\varepsilon$ 消掉，剩下\textbf{$\varepsilon$-自由卷积}
    \[
      a_0[j] = \textstyle\sum_t (\pm)\,R[(j-p_t)\bmod d]
    \]
    其对均匀挑战位置的期望
    \[
      \mathbb{E}[a_0[j]] = \frac{\tau}{d}\bigl(2\,\mathrm{CumR}[j]-\Sigma R\bigr)

    \column{0.40\textwidth}
    \begin{exampleblock}{传播到 $\Phi$}

      \beta (KR)[m]\bigr] + \cdots$\\[0.2em]
      $K[m,v] = d - 2|m-v|$（tent 核）\\
      $\beta = N\tau(\tau-1)/d^2 = 1.33\times10^4$

  \end{columns}

  \begin{block}{最重要的一句话}
    \small 场预测与实测残差 corr 0.967--0.997。

    \item \textbf{X6 跨数据集}：real-1M 与 sim3m 各自独立拟合的 $\hat R$
          逐坐标一致 761/768（不一致的恰是 1M 时的 7 个噪声跨界），corr 0.999。
    \item \textbf{X3 管线完整性}：sim3m 全部 300 万条签名经原始验证器重验证零失败；
          \textbf{X8}：独立复现 $z_1$ 回归路线，同一把等价私钥。
  \end{enumerate}
  \begin{center}
    {\color{ravbronze}\small 信任，但要用七种不同的方式验证。}
  \end{center}
\end{frame}

\begin{frame}{与外部世界对表}
  \begin{itemize}
    \item \textbf{同期的独立分析}提出了基于 $z_1$ 回归的同族路线。
          本工作\textbf{独立见证并已完整复现该路线}（RMSE 0.1842@200k vs 原文
          0.1863；闭合后同一把等价私钥）；\textbf{我们的路线不同}：
          hint 通道（tent 场算子求逆），$4\times10^6$ 零错误闭合。
        \item \textbf{ePrint 2013/383}（Ducas 等）：双峰 Fiat--Shamir 的奠基。
          \textbf{ePrint 2017/505}（Espitau 等，CCS 2017）：BLISS 双峰拒绝
          采样的缓存侧信道 + EM——\textbf{需要运行时泄露}；我们的路线
          \textbf{纯公开数据}。
  \end{itemize}
\end{frame}

\begin{frame}{怎么修：给拉文克劳的三个方案}
  \begin{enumerate}
    \item \textbf{别再共享符号位}：每次签名（甚至每坐标）独立掷
          $\varepsilon$——同时掐断 $z_0$ 标签与 hint 的 $\varepsilon$-对齐前提。
    \item \textbf{加宽 $z_0$ 掩码}：$\gamma_{1,0}=8$ 的支撑错位太"好读"；
          加宽可稀释标签（但 hint 的 $a$ 观测不受此影响——单修不够）。
    \item \textbf{重审 hint 论证式}："$w-z_2+b_0*c$" 让掩码抵消、秘密一阶线性
          进入公开值。任何"验证器可重算 + 一次一密桥接"的设计都要按
          \textbf{响应-秘密的独立性与可逆性}重审。
  \end{enumerate}
  \begin{center}
    {\color{ravbronze}\small 最好的防御者，是知道攻击者如何思考的人。}
  \end{center}
\end{frame}

\begin{frame}{方法论：拉文克劳的四条家规}
  \begin{enumerate}
    \item \textbf{预注册阈值}：判决阈值在看到数据前由公开常数合成——
          不给"调参到收口"留门。
    \item \textbf{三臂对照}：oracle / attack / shuffled（打乱标签必须塌回基线）。
    \item \textbf{三级证据}：纯数学推导 / 理想化模型 / 真实 C 闭环——
          每个数字标注属于哪一级。
    \item \textbf{交叉验证}：不同数据集组合、不同代码路径、不同轮次——
          同一结论要能被独立地重新得到。
  \end{enumerate}
  \begin{exampleblock}{否定结果与更正也是资产}
    \small $\psi$ 统计量伪象已更正、ICM 发散、EM 不可辨识、e-场被否——
    "此路不通"砍掉一半搜索空间；$z_1$ 回归路线的独立复现又为两条路线提供交叉见证。
  \end{exampleblock}
\end{frame}
